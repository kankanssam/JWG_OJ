구글 스프레드시트 DB 연동 설정

                
            

            

                
⚠️ 중요: 독립된 스프레드시트 연동을 위해 기존의 getActiveSpreadsheet() 대신 지정된 시트 ID 강제 오픈 코드로 템플릿을 수정했습니다. 아래 코드를 전체 교체하여 Apps Script에 저장하십시오.

                

                    
                    
                    
		

                    
 관리자 권한 부여 (소유자 전용)

                    
지정할 사용자의 ID를 입력하면 해당 사용자는 관리자 등급이 되어 연동 설정을 변경할 수 있습니다.

// 구글 앱스 스크립트(Apps Script) 독립 실행형 안정화 템플릿
var SPREADSHEET_ID = "1xMwqXvcksPdwKKnoKXJB1CM9wHeFI-oHeK4lkKoyjg8";

function getTargetSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch(e) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  var action = e.parameter.action;
  var ss = getTargetSpreadsheet();
  
  if (action === "getProblems") {
    var sheet = ss.getSheetByName("problems");
    var data = sheet.getDataRange().getValues();
    var list = [];
    if(data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        list.push({ id: data[i][0], title: data[i][1], source: data[i][2], category: data[i][3], description: data[i][4], inputSpec: data[i][5], outputSpec: data[i][6], sampleInput: data[i][7], sampleOutput: data[i][8], hiddenInput: data[i][9], hiddenOutput: data[i][10] });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(list)).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "getSubmissions") {
    var sheet = ss.getSheetByName("submissions");
    var data = sheet.getDataRange().getValues();
    var list = [];
    if(data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        list.push({ submissionId: data[i][0], timestamp: data[i][1], userId: data[i][2], problemId: data[i][3], language: data[i][4], status: data[i][6], resultMessage: data[i][7] });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(list)).setMimeType(ContentService.MimeType.JSON);
  }				
}

function doPost(e) {
  var ss = getTargetSpreadsheet();
  var params = JSON.parse(e.postData.contents);
  var action = params.action;
  
  if (action === "register") {
    var sheet = ss.getSheetByName("users");
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] == params.userId) {
        return ContentService.createTextOutput(JSON.stringify({success: false, message: "이미 존재하는 ID입니다."})).setMimeType(ContentService.MimeType.JSON);
      }
    }
    sheet.appendRow([params.userId, params.password, params.school, params.role, params.name, 0]);
    return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "login") {
    var sheet = ss.getSheetByName("users");
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] == params.userId && data[i][1] == params.password) {
        return ContentService.createTextOutput(JSON.stringify({ success: true, user: { userId: data[i][0], school: data[i][2], role: data[i][3], name: data[i][4], score: Number(data[i][5]) } })).setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({success: false, message: "ID 또는 비밀번호가 일치하지 않습니다."})).setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "submit") {
    var sheet = ss.getSheetByName("submissions");
    var subId = "SUB-" + Math.floor(Math.random() * 90000 + 10000);
    sheet.appendRow([subId, new Date(), params.userId, params.problemId, params.language, params.code, params.status, params.resultMessage]);
    
    // Update Score if Correct
    if(params.status === "Correct") {
       var uSheet = ss.getSheetByName("users");
       var uData = uSheet.getDataRange().getValues();
       for (var j = 1; j < uData.length; j++) {
          if(uData[j][0] == params.userId) {
             uSheet.getRange(j+1, 6).setValue(Number(uData[j][5]) + 10);
             break;
          }
       }
    }
    return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
  }
}
